\begin{frame}{Motivation:  Clusters or Community}
\begin{itemize}
\item Unsupervised Machine Learning Technique
\item Networks
\item Non-Overlapping Partitions
\item Competing Objectives
\begin{itemize}
\item Maximize Friends
\item Minimize Strangers
\end{itemize} 
\item Frustrated Choice (Cannot Max/Min both)
\end{itemize}
\end{frame}

\begin{frame}{Background:  Robustness}
\begin{itemize}
\item A node is {\bf robust} if its current community is unambiguously preferred over all other communities.

\item For a given partition, the robustness will be the fraction of nodes that there is no better move for.
\end{itemize}
\end{frame}
\begin{frame}{Background:  Familiarity Index}
\[
F_i(C_A,C_B) = \frac{\Delta d}{\Delta d + \Delta \hat d}
\]
\begin{itemize}
\item $F>1$ or $F<0$ you should make that change
\item $0<F<1$ are frustrated choices
\end{itemize}
\end{frame}

\begin{frame}{Background:  Constant Potts Model}
Comes from an analog of the Hamiltonian for the magnetic spin in statistical mechanics.
\[
v_{ij} = \left(A_{ij} - \gamma\right)\delta_{v_i,v_j\in C_k}
\]

Can vary $\gamma$:
\begin{itemize}
\item $\gamma = 0$ ignores strangers
\item $\gamma = 1$ ignores friends
\item Frustrated nodes 
\begin{itemize}
\item $\gamma<F_i$ go to more friends
\item $\gamma>F_i$ go to less strangers
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Methodology:  Game Theory}
\begin{itemize}
\item Agents (nodes here) act selfishly
\item Cooperation happens (stable communities emerge)
\item Nash Equilibrium happens
\item Equilibrium is beneficial for collective
\end{itemize}
\end{frame}
\begin{frame}{Methodology:  Hedonism}
\begin{itemize}
\item Hedonic Potential created
\item Utility Functions can be computed in polynomial time
\item Three scales:
\begin{enumerate}
\item Node
\[
\varphi^\gamma_i\left(C_k\right) = \sum_{j\in C_k}v_{ij}
\]
\item Community
\[
\phi^\gamma\left(C_k\right) = \frac12\sum_{i\in C_k}\sum_{j\in C_k}v_{ij}
\]
\item Partition
\[
\Phi^\gamma(\pi) = \sum_{k=1}^K\phi^{\gamma}\left(C_k\right) = \left(1-\gamma\right)\sum_{k=1}^K m_k - \gamma\sum_{k=1}^K\left(\binom{n_k}2 - m_k\right)
\]
\end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Methodology:  Leiden Algorithm}
\begin{enumerate}
\item Local Moving of Nodes:  Uses a random queue and only looks at moving neighbors after evaluating a node
\item Refinement or Partition:  Look into a community and makes random but improving choices (not greedy!)
\item Aggregation of Network:  Take the refinements and reorganize them into communities  
\end{enumerate}
\end{frame}

\begin{frame}{Methodology:  Game Equilibrium}
\begin{itemize}
\item Stability (or Equilibrium)
\begin{itemize}
\item Internally Stable Communities:  No node inside a community has incentive to leave
\item Externally Stable Communities:  No node outside community has incentive to join
\item Locally Stable Node: Nodes do not have incentive to deviate
\end{itemize}
\item Solution is partition with all nodes locally stable
\end{itemize}
\end{frame}
 
\begin{frame}{Results and Findings: Overview}
\begin{itemize}
\item $\gamma$ is rational
\item Guaranteed convergence 
\item pseudo-polynomial time
\end{itemize}

Why?
\begin{itemize}
\item $\Phi$ is bounded within $[-n^2,n^2]$ and increasing
\item Since $\gamma = \dfrac bc$, each beneficial node increases potential by $\dfrac1c$.
\item Convergence happens in $O(cn^2)$
\end{itemize}
\end{frame}
